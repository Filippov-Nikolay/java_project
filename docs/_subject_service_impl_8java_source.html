94 @Override

98 if (dto.getTeacherId() == null || dto.getSubjectId() == null || dto.getGroupId() == null) {
99 log.error("Assignment failed: One of the IDs is null. DTO: {}", dto);
100 throw new IllegalArgumentException("Вчитель, предмет та група мають бути обрані");

102
103 User teacher = userRepository.findById(dto.getTeacherId())
104 .orElseThrow(() -> new NotFoundException("Вчитель", dto.getTeacherId()));
105
106 if (teacher.getRole() != Role.TEACHER) {
107 throw new IllegalArgumentException("Користувач не є викладачем");
108 }
109
110 Subject subject = subjectRepository.findById(dto.getSubjectId())
111 .orElseThrow(() -> new NotFoundException("Предмет", dto.getSubjectId()));
112
113 Group group = groupRepository.findById(dto.getGroupId())
114 .orElseThrow(() -> new NotFoundException("Група", dto.getGroupId()));
115
116 // Додаємо зв'язки
117 teacher.getSubjects().add(subject);
118 group.getSubjects().add(subject);
119
120 userRepository.save(teacher);
121 groupRepository.save(group);
122
123 log.info("Successfully assigned teacher {} to subject {} for group {}",
124 teacher.getLastName(), subject.getName(), group.getName());
125 }
