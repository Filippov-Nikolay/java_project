30 @Override
31 @Transactional
32 public List<JournalRecord> getJournalBySchedule(Long scheduleId) {
33 log.info("Fetching journal records for schedule ID: {}", scheduleId);
34
35 // 1. Шукаємо існуючі записи в базі
36 List<JournalRecord> existingRecords = journalRepository.findByScheduleIdOrderByStudentFullNameAsc(scheduleId);
37
38 if (!existingRecords.isEmpty()) {
39 // Оновлюємо ПІБ студентів (на випадок, якщо вони змінилися в профілі)
40 existingRecords.forEach(record ->
41 userRepository.findById(record.getStudentId())
42 .ifPresent(u -> record.setStudentFullName(u.getLastName() + " " + u.getFirstName()))
43 );
44 return existingRecords;
45 }
46
47 // 2. Якщо записів немає, створюємо їх на основі активних студентів групи
48 Schedule schedule = scheduleRepository.findById(scheduleId)
49 .orElseThrow(() -> new NotFoundException("Заняття", scheduleId));
50
51 log.info("First time opening journal for group: {}. Generating student list.", schedule.getGroup().getName());
52
53 List<User> students = userRepository.findByGroupIdAndRoleAndEnabledTrueOrderByLastNameAsc(
54 schedule.getGroup().getId(),
55 Role.STUDENT
56 );
57
58 return students.stream().map(student -> JournalRecord.builder()
59 .scheduleId(scheduleId)
60 .studentId(student.getId())
61 .studentFullName(student.getLastName() + " " + student.getFirstName())
62 .attendance(null)
63 .grade("")
64 .workType(WorkType.REGULAR)
65 .build()
66 ).collect(Collectors.toList());
67 }
