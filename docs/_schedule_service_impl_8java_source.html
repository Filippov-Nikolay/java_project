120 @Override
121 @Transactional
122 public void importFromExcel(MultipartFile file) {
123 try (InputStream is = file.getInputStream();
124 Workbook workbook = new XSSFWorkbook(is)) {
125
126 Sheet sheet = workbook.getSheetAt(0);
127 DataFormatter formatter = new DataFormatter();
128
129 for (int i = 1; i <= sheet.getLastRowNum(); i++) {
130 Row row = sheet.getRow(i);
131 if (row == null || row.getCell(0) == null) continue;
132
133 final int currentRowNum = i + 1;
134
135 String dateStr = formatter.formatCellValue(row.getCell(0));
136 String lessonNumStr = formatter.formatCellValue(row.getCell(1));
137 String subjectName = formatter.formatCellValue(row.getCell(2));
138 String teacherFullName = formatter.formatCellValue(row.getCell(3));
139 String groupName = formatter.formatCellValue(row.getCell(4));
140 String typeStr = formatter.formatCellValue(row.getCell(5));
141 String room = formatter.formatCellValue(row.getCell(6));
142
143 int lessonNum = Integer.parseInt(lessonNumStr.replaceAll("[^0-9]", ""));
144
145 Group group = groupRepository.findByName(groupName)
146 .orElseThrow(() -> new RuntimeException("Рядок " + currentRowNum + ": Група '" + groupName + "' не знайдена"));
147
148 Subject subject = subjectRepository.findByName(subjectName)
149 .orElseThrow(() -> new RuntimeException("Рядок " + currentRowNum + ": Предмет '" + subjectName + "' не знайдений"));
150
151 String[] nameParts = teacherFullName.split(" ");
152 if (nameParts.length < 2) {
153 throw new RuntimeException("Рядок " + currentRowNum + ": Некоректне ПІБ вчителя '" + teacherFullName + "'");
154 }
155
156 // ВИПРАВЛЕНО: Викликаємо метод, який реально існує в репозиторії
157 User teacher = userRepository.findByLastNameAndFirstNameAndEnabledTrue(nameParts[0], nameParts[1])
158 .orElseThrow(() -> new RuntimeException("Рядок " + currentRowNum + ": Вчитель '" + teacherFullName + "' не знайдений або деактивований"));
159
160 Schedule schedule = Schedule.builder()
161 .date(LocalDate.parse(dateStr))
162 .lessonNumber(lessonNum)
163 .group(group)
164 .subject(subject)
165 .teacher(teacher)
166 .type(LessonType.valueOf(typeStr.toUpperCase()))
167 .room(room)
168 .build();
169
170 scheduleRepository.save(schedule);
171 }
172 log.info("Excel import completed successfully");
173 } catch (Exception e) {
174 log.error("Excel import failed: {}", e.getMessage());
175 throw new RuntimeException("Помилка імпорту: " + e.getMessage());
176 }
